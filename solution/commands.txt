git init - создать репозиторий
git add . - добавить файлы в индекс
git status - посмотреть статус репозитория
git commit -m - создать коммит
git push -u origin - отправить изменения на удаленный репозиторий
git remote add origin <Ссылка на удалённый репозиторий> - привязать удаленный репозиторий
git remote remove origin - отвязать удаленный репозиторий
git log - посмотреть коммиты
git pull - скачивает изменения из удаленного репозитория
